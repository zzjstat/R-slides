\begin{frame}[fragile]{上讲回顾：变量与类型}
  \small
  变量保存输入值或计算结果。下面定义三种基本类型的变量。
  \inputlabel
\begin{lstlisting}
n <- 23
name <- "My"
flag <- TRUE
\end{lstlisting}
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}llll@{}}
    \toprule
    \thead{变量} & \thead{class() 返回值} & \thead{检查代码} & \thead{结果} \\
    \midrule
    \code{n} & \code{"numeric"} & \code{is.numeric(n)} & \code{TRUE} \\
    \code{name} & \code{"character"} & \code{is.character(name)} & \code{TRUE} \\
    \code{flag} & \code{"logical"} & \code{is.logical(flag)} & \code{TRUE} \\
    \bottomrule
  \end{tabular*}\par
  \vspace{0.2em}
  \code{class()} 查看类型，类型检查函数返回 \code{TRUE} 或 \code{FALSE}。\par
  数值计算中，\code{TRUE} 按 1 计算，\code{FALSE} 按 0 计算。
  \note{教材第18--19页。与下一页合计复习3分钟。沿用第一讲已学内容，不重新展开类型和常量。}

\begin{frame}{上讲回顾：常量与命名}
  \small
  常量是直接写在代码中的值。以下系统常量有特定含义。
  \par\vspace{0.4em}
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}lp{0.73\linewidth}@{}}
    \toprule
    \thead{名称} & \thead{含义} \\
    \midrule
    \code{pi} & 圆周率，约为 3.141593 \\
    \code{Inf} & 正无穷，例如 \code{1 / 0} 的结果 \\
    \code{NaN} & 无法定义的数值结果，例如 \code{0 / 0} \\
    \code{NA} & 缺失值，例如没有记录的观测值 \\
    \bottomrule
  \end{tabular*}\par
  \begin{itemize}
    \item 名称可含字母、数字、下划线和句点；\\
          区分大小写，不以数字开头。
    \item 本课统一用 \code{<-} 赋值。\code{=} 也可赋值，\code{\#} 用于注释。
    \item 直接输入变量名可查看结果；不覆盖已有函数或常量。
  \end{itemize}
  \note{教材第16--19页。常量、变量名及查看结果均为第一讲复习。字符值用英文引号，逻辑值TRUE和FALSE用大写。}
\end{frame}

\begin{frame}[fragile]{组合向量}
  \code{c()} 可以组合多个值，也可以组合已有向量。\par
  单个数值也可看作只有一个元素的数值向量。
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.62\textwidth}
  \inputlabel
\begin{lstlisting}
x1 <- c(1, 2)
x2 <- c(3, 4)
vec <- c(x1, x2)
vec
\end{lstlisting}
    \end{column}
    \begin{column}{0.34\textwidth}
  \outputlabel
\begin{lstlisting}[style=output]
[1] 1 2 3 4
\end{lstlisting}
    \end{column}
  \end{columns}
  \par\vspace{0.35em}
  \begin{itemize}
    \item 本例按 \code{x1}、\code{x2} 的顺序组成新向量。
    \item 字符型或逻辑型向量也使用 \code{c()} 创建。
    \item 不用 \code{c} 作为保存数据的变量名。
  \end{itemize}
  \note{教材第18--20页的c()和基本向量。向量部分10分钟（第一节第5--15分钟），复习c()后转入seq()和多个正整数下标。仅使用同类型向量。}
\end{frame}

\begin{frame}{seq() 的常用参数}
  \small
  \code{seq()} 用于生成等差数列。
  \par\vspace{0.5em}
  \renewcommand{\arraystretch}{1.25}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}ll@{}}
    \toprule
    \thead{参数} & \thead{含义} \\
    \midrule
    \code{from} & 起始值 \\
    \code{to} & 终止值 \\
    \code{by} & 步长，即相邻两个值之差 \\
    \code{length.out} & 生成的值的个数 \\
    \bottomrule
  \end{tabular*}\par
  \vspace{0.6em}
  教材使用以下三种组合：
  \begin{enumerate}
    \item 起始值、终止值、步长。
    \item 起始值、终止值、个数。
    \item 起始值、步长、个数。
  \end{enumerate}
  \note{教材第21页的三种seq()调用。只介绍本例使用的参数组合，不要求同时填写全部参数。}
\end{frame}

\begin{frame}[fragile]{按步长生成序列}
  \small
  \code{by} 指定相邻两个值之差。
  \inputlabel
\begin{lstlisting}
seq(-2, 3, by = 0.5)
seq(10, 2, by = -2)
\end{lstlisting}
  \outputlabel
\begin{lstlisting}[style=output]
[1] -2.0 -1.5 -1.0 -0.5 0.0 0.5 1.0
[8] 1.5 2.0 2.5 3.0
[1] 10 8 6 4 2
\end{lstlisting}
  \begin{itemize}
    \item 第一行从 $-2$ 开始，每次增加 0.5，到 3 结束。
    \item 第二行每次减少 2，得到递减序列。
  \end{itemize}
  \note{教材第21页的seq(-2,3,by=0.5)。递减例沿用原课件，说明步长参数取负值的同一操作。输出换行与Console宽度有关，行首编号表示该行第一个元素的位置。}
\end{frame}

\begin{frame}[fragile]{按个数生成序列}
  指定起始值、终止值和个数，由 R 确定步长。
  \inputlabel
\begin{lstlisting}
seq(2, 4, length.out = 4)
\end{lstlisting}
  \outputlabel
\begin{lstlisting}[style=output]
[1] 2.000000 2.666667 3.333333 4.000000
\end{lstlisting}
  \begin{itemize}
    \item 生成 4 个值，首项为 2，末项为 4。
    \item 相邻值之间的差相同。
  \end{itemize}
  \note{教材第21页的seq(2,4,length.out=4)。这里的length.out是参数名，不引入length()函数。}
\end{frame}

\begin{frame}[fragile]{同时指定步长和个数}
  起始值、步长和个数确定后，末项也随之确定。
  \inputlabel
\begin{lstlisting}
seq(2, by = 0.4, length.out = 4)
\end{lstlisting}
  \outputlabel
\begin{lstlisting}[style=output]
[1] 2.0 2.4 2.8 3.2
\end{lstlisting}
  \begin{itemize}
    \item 从 2 开始，每次增加 0.4，共生成 4 个值。
    \item 本例不再另外指定 \code{to}。
  \end{itemize}
  \note{教材第21页的seq(2,by=0.4,length=4)，采用完整参数名length.out，避免将部分匹配作为新知识介绍。}
\end{frame}

\begin{frame}[fragile]{提取向量中的多个元素}

    \item \code{vec[c(1, 3)]} 依次提取第 1、3 个元素。
    \item 位置编号从 1 开始。本讲使用已有元素位置\\
          \mbox{作为正整数下标}，本例为 1、2、3。

  \note{教材第21--22页的正整数下标。单下标为复习，多下标为本讲新内容。}

\begin{frame}[fragile]{修改向量中的多个元素}

  \begin{itemize}

    \item 第二行将第 1、3 个元素分别改为 11、13。
    \item 替换值的顺序与所选位置的顺序一致。
  \end{itemize}
  \note{教材第22页的下标赋值。多个位置的修改沿用原课件，只使用正整数位置和等长替换值，不展开其他索引。}

\begin{frame}{矩阵的行与列}
  \small
  矩阵按行、列排列数据，各元素具有相同的数据类型。
  \par\vspace{0.6em}
  \renewcommand{\arraystretch}{1.6}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}lccc@{}}
    \toprule
    & \thead{第1列} & \thead{第2列} & \thead{第3列} \\
    \midrule
    \thead{第1行} & 1 & 5 & 9 \\
    \thead{第2行} & 3 & 7 & 11 \\
    \bottomrule
  \end{tabular*}\par
  \vspace{0.6em}
  \begin{itemize}
    \item 本例是 2 行、3 列的数值型矩阵。
    \item 行号和列号共同确定一个元素的位置。
    \item 向量只有元素顺序，矩阵还具有行、列结构。
  \end{itemize}
  \note{教材第22页的矩阵定义和2行3列示例。矩阵讲解17分钟（第一节第15--32分钟），以教师演示为主。第19--22页讲清位置选择与修改结果，不涉及矩阵乘法或求逆。}
\end{frame}

\begin{frame}[fragile]{创建矩阵：默认按列填充}
  \small
  \code{matrix()} 将向量中的值排列成矩阵。
  \inputlabel
\begin{lstlisting}
m <- seq(1, 11, by = 2)
A <- matrix(m, nrow = 2, ncol = 3)
A
\end{lstlisting}
  \outputlabel
\begin{lstlisting}[style=output]
     [,1] [,2] [,3]
[1,]    1    5    9
[2,]    3    7   11
\end{lstlisting}
  \begin{itemize}
    \item \code{nrow} 指定行数，\code{ncol} 指定列数。
    \item 默认先填满第 1 列，再填第 2、3 列。
  \end{itemize}
  \note{教材第22页：x<-seq(1,11,by=2)和matrix(x,2,3)，此处写出参数名便于辨认。}
\end{frame}

\begin{frame}[fragile]{创建矩阵：按行填充}
  \small
  设置 \code{byrow = TRUE}，先填满一行再填下一行。
  \inputlabel
\begin{lstlisting}
B <- matrix(1:6, nrow = 2, ncol = 3,
            byrow = TRUE)
B
\end{lstlisting}
  \outputlabel
\begin{lstlisting}[style=output]
     [,1] [,2] [,3]
[1,]    1    2    3
[2,]    4    5    6
\end{lstlisting}
  \begin{itemize}
    \item 数据、行数和列数已确定，填充顺序影响元素位置。
    \item \code{TRUE} 是已学过的逻辑值。
  \end{itemize}
  \note{教材第23页matrix(1:6,2,3,byrow=TRUE)。采用完整TRUE，不扩展简写T。}
\end{frame}

\begin{frame}{矩阵下标的写法}
  \small
  方括号内，逗号前选择行，逗号后选择列。
  \par\vspace{0.5em}
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}ll@{}}
    \toprule
    \thead{代码} & \thead{提取内容} \\
    \midrule
    \code{A[1, 2]} & 第1行、第2列的一个元素 \\
    \code{A[1, ]} & 第1行的全部元素 \\
    \code{A[, 1]} & 第1列的全部元素 \\
    \code{A[c(1, 2), c(1, 3)]} & 指定行、列组成的子矩阵 \\
    \bottomrule
  \end{tabular*}\par
  \vspace{0.6em}
  空白表示保留该方向上的全部位置。\par
  行号、列号均从 1 开始。\par
  同一写法可放在 \code{<-} 左侧，修改元素、整行、整列或子矩阵。
  \note{教材第23页：a[i,j]、a[i,]、a[,j]。这里只使用正整数下标，不讲drop等参数。}
\end{frame}

\begin{frame}[fragile]{提取子矩阵}
