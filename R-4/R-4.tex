\begin{frame}{条件语句的三种形式}
  条件语句根据判断结果决定执行哪段代码。
  \par\vspace{0.4em}
  \renewcommand{\arraystretch}{1.45}
  \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}ll@{}}
    \toprule
    \thead{形式} & \thead{执行规则} \\
    \midrule
    \code{if} & 条件成立时执行 \\
    \code{if / else} & 两种情形中执行一种 \\
    \code{if / else if / else} & 依次判断，执行相应分支 \\
    \bottomrule
  \end{tabular*}
  \par\vspace{0.5em}
  \code{if} 的条件应得到一个 \code{TRUE} 或 \code{FALSE}。
  \note{教材第99--101页。条件语句部分合计15分钟（第一节第3--18分钟）。本页1分钟，三种形式均用单一数值作判断。}
\end{frame}

\begin{frame}[fragile]{单条件语句：\texttt{if}}
  \small
  条件不成立时跳过花括号中的代码；成立时执行。
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.65\textwidth}
      \inputlabel
\begin{lstlisting}
x <- 1
if (x < 1) {
  x <- x + 1
}
x
if (x < 2) {
  x <- x + 1
}
x
\end{lstlisting}
    \end{column}
    \begin{column}{0.30\textwidth}
      \outputlabel
\begin{lstlisting}[style=output]
[1] 1
[1] 2
\end{lstlisting}
      \small 第一处不执行。\par 第二处执行。
    \end{column}
  \end{columns}
  \note{教材第99页原例。2分钟，跟踪同一变量x从1到2的变化。两次输出分别对应两处单独的x。}
\end{frame}

\begin{frame}[fragile]{两种情形：\texttt{if / else}}
  \small
  条件成立时执行前一分支，否则执行后一分支。
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.65\textwidth}
      \inputlabel
\begin{lstlisting}
x <- 1
if (x < 1) {
  x <- x - 1
} else {
  x <- x + 1
}
x
\end{lstlisting}
    \end{column}
    \begin{column}{0.30\textwidth}
      \outputlabel
\begin{lstlisting}[style=output]
[1] 2
\end{lstlisting}
    \end{column}
  \end{columns}
  \par\smallskip
  \code{else} 与前一分支的右花括号写在同一行。
  \note{教材第99--100页。2分钟，沿用x初值为1的二选一结构；强调本例只会执行一个分支。}
\end{frame}

\begin{frame}{教材例 2.A.3：计算分布函数}
  对给定的一个数值 $x$，计算
  \[
    F(x)=\begin{cases}
      0, & x\leq 0,\\[0.3em]
      x, & 0<x\leq 1,\\[0.3em]
      1, & x>1.
    \end{cases}
  \]
  \begin{itemize}
    \item 先判断 \code{x <= 0}。
    \item 该条件不成立后，再判断 \code{x <= 1}。
    \item 前两个条件都不成立时，取值为 1。
  \end{itemize}
  \note{教材第101页例2.A.3。2分钟，联系数学分段和代码分支；说明第二次判断前已知x>0，无需再写重复条件。}
\end{frame}

\begin{frame}{\texttt{for} 循环：依次取值}
  \code{for} 循环让循环变量依次取向量中的值，\par
  每取一个值就执行一次循环体。
  \par\vspace{0.5em}
  \begin{enumerate}
    \item 取向量中的第一个值。
    \item 执行花括号中的程序代码。
    \item 取下一个值，继续执行，直到向量中的值取完。
  \end{enumerate}
  \note{教材第101--102页。for部分合计12分钟（第一节第18--30分钟）。本页1分钟，先讲执行过程，后讲可手算代码。}
\end{frame}

\begin{frame}[fragile]{骰子频率：循环完成重复步骤}
  \inputlabel
\begin{lstlisting}
for (i in 1:12) {
  if (i < 11) { n <- i * 10
  } else if (i == 11) { n <- 500
  } else { n <- 1000 }
  y <- x[1:n]
  f <- c(f, c(sum(y == 1), sum(y == 2),
              sum(y == 3), sum(y == 4),
              sum(y == 5), sum(y == 6)) / n)
}
\end{lstlisting}
  \small 每轮确定 \code{n}，提取前 \code{n} 次，再保存六种频率。
  \note{教材第103页原程序整理，接上一页x和f运行。3分钟教师演示，不要求学生从零重写；只逐步指出定n、取前n次、算六种频率三步。}
\end{frame}
