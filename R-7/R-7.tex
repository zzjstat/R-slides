\begin{frame}[fragile]{样本均值与中位数}
  均值概括数据的平均水平；\par
  中位数由排序后中间的数值确定。\par
  偶数个观测取中间两个数的平均。
\begin{lstlisting}
mean(score87)
median(score87)
\end{lstlisting}
\outputlabel
\begin{lstlisting}[style=output]
[1] 79.6
[1] 83.5
\end{lstlisting}
  用 \code{sort(score87)} 查看排序后的数据。
  \note{教材第153--156页。第一节10--13分钟，教师演示排序并指出中间两个数。}
\end{frame}
\begin{frame}[fragile]{观察一个数值的影响}

\begin{lstlisting}[style=output]

[1] 82

\begin{frame}[fragile]{众数与频数}
  众数是出现频数最高的取值。
\begin{lstlisting}
table(jia)
\end{lstlisting}
  甲班成绩中85出现4次，频数最高。
  \par\medskip
  若多个取值的频数并列最高，可以有多个众数。
  \par\medskip
  连续数据可用频率直方图最高柱对应组的\\
  组中点近似众数。
  \note{教材第157--158页。第一节16--20分钟。连续数据可用频率直方图最高柱对应组的组中点作近似；不把组中点说成精确众数。}
\end{frame}
\begin{frame}[fragile]{方差与标准差}
  方差、标准差描述数据的离散程度。
\begin{lstlisting}
var(jia)
sd(jia)
var(yi)
sd(yi)
\end{lstlisting}
  样本标准差是样本方差的平方根。
  \note{教材第159--161页。第一节20--24分钟，教师演示两班的结果。}
\end{frame}
\begin{frame}[fragile]{标准化}
  标准化分数表示观测值相对样本均值的位置。
\[
  z_i=\frac{x_i-\bar{x}}{s}
\]
\begin{lstlisting}
scale(jia)
\end{lstlisting}
  样本标准差应不为0。
  \note{教材第161--162页。第一节24--27分钟，展示部分输出，说明同一数据的相对位置。}
\end{frame}
\begin{frame}[fragile]{变异系数}
  变异系数比较相对离散程度：
\[
  \frac{s}{|\bar{x}|}
\]
\begin{lstlisting}
sd(jia) / abs(mean(jia))
\end{lstlisting}
  这里要求样本均值不为0。
  \note{教材第162--163页。第一节27--30分钟，只说明相对离散，不增加其他指标。}
\end{frame}

\begin{frame}[fragile]{样本分位数}
  将数据从小到大排列，分位数描述样本中某个比例对应的位置。
\begin{lstlisting}
quantile(score87, 0.4, type = 5)
\end{lstlisting}
\outputlabel
\begin{lstlisting}[style=output]
40%
 82
\end{lstlisting}
  \note{教材第163--164页。第二节0--4分钟。第164页公式对应type=5，例题代码写type=4；本例两者均为82，说明不同算法可能不同，不展开九种方法。}
\end{frame}
\begin{frame}[fragile]{四分位数与五数概括}
  五数概括包括最小值、下四分位数、中位数、\par
  上四分位数和最大值。
\begin{lstlisting}
fivenum(score87)
\end{lstlisting}
\outputlabel
\begin{lstlisting}[style=output]

\begin{frame}{四分位距与离群值}
  四分位距是上、下四分位数的差：
\[
  \mathrm{IQR}=Q_3-Q_1.
\]
  教材采用的常见判断界限为：
\[
  Q_1-1.5\,\mathrm{IQR},\qquad Q_3+1.5\,\mathrm{IQR}.
\]
  超过界限的数据需要结合背景进一步检查。
  \note{教材第167--168页。第二节15--19分钟。不将离群值直接认定为错误。}
\end{frame}
\begin{frame}[fragile]{箱线图}
  箱体从下四分位数延伸到上四分位数，\par
  中间线表示中位数。
\begin{lstlisting}
boxplot(score87, ylab = "成绩")
\end{lstlisting}
  须线延伸到界限内最远的观测值；\par
  界限之外的点单独标示。
  \note{教材第168--170页。第二节19--22分钟，须线不是必然等于1.5倍四分位距。}
\end{frame}
